}); scene.remove(pivot); rotationGroup = []; dragStage = "IDLE"; }); function animate() { requestAnimationFrame(animate); renderer.render(scene, camera); } animate(); window.addEventListener("resize", () => { camera.aspect = window.innerWidth / window.innerHeight;
